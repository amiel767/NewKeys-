package com.soundstage.mixer.audio

import kotlinx.coroutines.flow.MutableSharedFlow
import kotlinx.coroutines.flow.SharedFlow
import kotlinx.coroutines.flow.asSharedFlow

/**
 * Horloge maître partagée par tous les composants rythmiques (métronome,
 * looper, indicateur de beats). Une seule source de vérité pour le BPM,
 * la signature, la subdivision et l'ancrage.
 *
 * Thread-safe. Aucune dépendance Android. Utilisable en test unitaire.
 *
 * Le temps est exprimé en nanosecondes depuis System.nanoTime().
 * L'ancrage `barAnchorNs` représente le début de la mesure 0 de référence.
 * Toutes les positions sont dérivées de (nowNs - barAnchorNs).
 */
class TransportClock {

    data class Signature(val beatsPerBar: Int, val beatUnit: Int) {
        companion object {
            val SUPPORTED: List<Signature> = listOf(
                Signature(4, 4), Signature(3, 4), Signature(2, 4),
                Signature(6, 8), Signature(9, 8), Signature(12, 8),
                Signature(5, 4), Signature(7, 8), Signature(5, 8), Signature(3, 8),
            )
            fun parse(s: String): Signature {
                val parts = s.split("/")
                if (parts.size != 2) return Signature(4, 4)
                val b = parts[0].toIntOrNull() ?: 4
                val u = parts[1].toIntOrNull() ?: 4
                return Signature(b, u)
            }
            fun default() = Signature(4, 4)
        }
        fun label(): String = "$beatsPerBar/$beatUnit"
    }

    enum class Subdivision(val subdivPerBeat: Int, val label: String) {
        Quarter(1, "1/4"),
        Eighth(2, "1/8"),
        Sixteenth(4, "1/16"),
        Triplet(3, "Triplet");
        companion object {
            fun parse(s: String): Subdivision = when (s) {
                "1/8" -> Eighth
                "1/16" -> Sixteenth
                "triplet", "Triplet" -> Triplet
                else -> Quarter
            }
        }
    }

    data class BeatPosition(
        val barIndex: Long,
        val beatInBar: Int,
        val subInBeat: Int,
        val fractionalInSub: Float,
    )

    @Volatile var bpm: Int = 120
        set(value) {
            val clamped = value.coerceIn(40, 240)
            if (clamped != field) {
                field = clamped
                reanchorFromNow()
            }
        }

    @Volatile var signature: Signature = Signature.default()
        set(value) {
            if (value != field) {
                field = value
                reanchorFromNow()
            }
        }

    @Volatile var subdivision: Subdivision = Subdivision.Quarter

    @Volatile private var barAnchorNs: Long = System.nanoTime()

    private val _tickFlow = MutableSharedFlow<TickEvent>(replay = 0, extraBufferCapacity = 64)
    val ticks: SharedFlow<TickEvent> = _tickFlow.asSharedFlow()

    data class TickEvent(
        val barIndex: Long,
        val beatInBar: Int,
        val subInBeat: Int,
        val isAccentStrong: Boolean,
        val isAccentMedium: Boolean,
        val timestampNs: Long,
    )

    fun beatDurationNs(): Long {
        val b = bpm.coerceIn(40, 240)
        return (60_000_000_000L / b.toLong())
    }

    fun barDurationNs(): Long = beatDurationNs() * signature.beatsPerBar.toLong()

    fun subDurationNs(): Long = beatDurationNs() / subdivision.subdivPerBeat.toLong()

    fun barAnchor(): Long = barAnchorNs

    fun currentPosition(): BeatPosition = positionAt(System.nanoTime())

    fun positionAt(nowNs: Long): BeatPosition {
        val barDur = barDurationNs().coerceAtLeast(1L)
        val subDur = subDurationNs().coerceAtLeast(1L)
        val elapsed = (nowNs - barAnchorNs).coerceAtLeast(0L)
        val barIndex = elapsed / barDur
        val inBar = elapsed % barDur
        val beatInBar = (inBar / beatDurationNs()).toInt().coerceIn(0, signature.beatsPerBar - 1)
        val inBeat = inBar % beatDurationNs()
        val subInBeat = (inBeat / subDur).toInt().coerceIn(0, subdivision.subdivPerBeat - 1)
        val frac = ((inBeat % subDur).toFloat() / subDur.toFloat()).coerceIn(0f, 1f)
        return BeatPosition(barIndex, beatInBar, subInBeat, frac)
    }

    fun timeToNextSubdivision(nowNs: Long = System.nanoTime()): Long {
        val subDur = subDurationNs()
        val elapsed = (nowNs - barAnchorNs).coerceAtLeast(0L)
        val next = ((elapsed / subDur) + 1) * subDur
        return (barAnchorNs + next) - nowNs
    }

    fun timeToNextBar(nowNs: Long = System.nanoTime()): Long {
        val barDur = barDurationNs()
        val elapsed = (nowNs - barAnchorNs).coerceAtLeast(0L)
        val next = ((elapsed / barDur) + 1) * barDur
        return (barAnchorNs + next) - nowNs
    }

    fun reanchorTo(nowNs: Long) {
        barAnchorNs = nowNs
    }

    private fun reanchorFromNow() {
        // Ré-ancrage précis : on aligne le prochain battement sur l'instant courant
        // pour éviter la dérive cumulative après un changement de BPM.
        val now = System.nanoTime()
        barAnchorNs = now
    }

    fun emitTickFromScheduler(bar: Long, beat: Int, sub: Int, strong: Boolean, medium: Boolean, ts: Long) {
        _tickFlow.tryEmit(TickEvent(bar, beat, sub, strong, medium, ts))
    }

    fun isStrongBeat(beatInBar: Int): Boolean = beatInBar == 0

    fun isMediumBeat(beatInBar: Int): Boolean {
        return when (signature.beatsPerBar) {
            2 -> beatInBar == 0
            3 -> beatInBar == 0
            4 -> beatInBar == 0 || beatInBar == 2
            5 -> beatInBar == 0 || beatInBar == 3
            6 -> beatInBar == 0 || beatInBar == 3
            7 -> beatInBar == 0 || beatInBar == 2 || beatInBar == 4
            9 -> beatInBar == 0 || beatInBar == 3 || beatInBar == 6
            12 -> beatInBar == 0 || beatInBar == 3 || beatInBar == 6 || beatInBar == 9
            else -> beatInBar == 0
        }
    }
}

package com.soundstage.mixer.audio

import android.media.AudioAttributes
import android.media.AudioFormat
import android.media.AudioManager
import android.media.AudioTrack
import android.os.Build
import android.util.Log
import kotlinx.coroutines.*
import kotlin.math.PI
import kotlin.math.exp
import kotlin.math.sin
import kotlin.math.pow

/**
 * Métronome autonome, synchronisé sur TransportClock.
 * Synthèse algorithmique des 6 sons (aucun asset).
 * Thread d'écriture dédié, AudioTrack statique en MODE_STATIC pour latence
 * minimale, réutilisation des buffers pré-calculés.
 *
 * Supporte : 10 signatures, 4 subdivisions, accent fort/moyen par signature,
 * volume global, click-in (count-in) au prochain début de mesure.
 */
class MetronomeEngine(private val clock: TransportClock) {

    private val TAG = "MetronomeEngine"
    private val sampleRate = 44100

    enum class Sound(val label: String) {
        Woodblock("Woodblock"),
        Beep("Beep"),
        Digital("Digital"),
        Click("Click"),
        Kick("Kick"),
        Conga("Conga");
        companion object {
            fun parse(s: String): Sound = values().firstOrNull { it.label.equals(s, true) } ?: Woodblock
        }
    }

    @Volatile var isActive: Boolean = false
        private set
    @Volatile var volume: Float = 0.65f
    @Volatile var accent1: Boolean = true
    @Volatile var accent13: Boolean = false
    @Volatile var visualFlash: Boolean = true
    @Volatile var currentSound: Sound = Sound.Woodblock

    private var audioTrack: AudioTrack? = null
    private var schedulerJob: Job? = null
    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())

    // Buffer PCM par son et par accent (fort / moyen / normal)
    private val bufferStrong: ShortArray by lazy { synthStrong(currentSound) }
    private val bufferMedium: ShortArray by lazy { synthMedium(currentSound) }
    private val bufferNormal: ShortArray by lazy { synthNormal(currentSound) }

    private var bufferedSound: Sound = Sound.Woodblock

    fun start() {
        if (isActive) return
        isActive = true
        bufferedSound = currentSound
        openTrack()
        schedulerJob = scope.launch { runScheduler() }
    }

    fun stop() {
        isActive = false
        schedulerJob?.cancel()
        schedulerJob = null
        try {
            audioTrack?.pause(); audioTrack?.flush(); audioTrack?.stop(); audioTrack?.release()
        } catch (_: Exception) {}
        audioTrack = null
    }

    fun release() {
        stop()
        scope.cancel()
    }

    fun changeSound(s: Sound) {
        if (bufferedSound == s) return
        currentSound = s
        // La reconstruction lazy se fait au prochain start()
        if (isActive) {
            stop(); start()
        }
    }

    private fun openTrack() {
        try {
            val minBuf = AudioTrack.getMinBufferSize(
                sampleRate, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT
            )
            val builder = AudioTrack.Builder()
                .setAudioAttributes(
                    AudioAttributes.Builder()
                        .setUsage(AudioAttributes.USAGE_MEDIA)
                        .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION)
                        .build()
                )
                .setAudioFormat(
                    AudioFormat.Builder()
                        .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                        .setSampleRate(sampleRate)
                        .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                        .build()
                )
                .setBufferSizeInBytes(max(minBuf, 4096))
                .setTransferMode(AudioTrack.MODE_STREAM)
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                builder.setPerformanceMode(AudioTrack.PERFORMANCE_MODE_LOW_LATENCY)
            }
            val track = builder.build()
            if (track.state == AudioTrack.STATE_INITIALIZED) {
                track.play()
                audioTrack = track
            } else {
                Log.w(TAG, "AudioTrack failed to init")
                track.release()
            }
        } catch (e: Exception) {
            Log.e(TAG, "openTrack: ${e.message}")
        }
    }

    private suspend fun runScheduler() {
        while (isActive) {
            val nowNs = System.nanoTime()
            val pos = clock.positionAt(nowNs)
            val subNs = clock.subDurationNs()
            val nextSubInNs = clock.timeToNextSubdivision(nowNs)
            val targetNs = nowNs + nextSubInNs

            val nextSub = (pos.subInBeat + 1) % clock.subdivision.subdivPerBeat
            val nextBeatInBar = if (nextSub == 0) (pos.beatInBar + 1) % clock.signature.beatsPerBar else pos.beatInBar
            val nextBar = if (nextSub == 0 && nextBeatInBar == 0) pos.barIndex + 1 else pos.barIndex

            val strong = nextSub == 0 && nextBeatInBar == 0 && accent1
            val medium = nextSub == 0 && clock.isMediumBeat(nextBeatInBar) && (accent13 || clock.isMediumBeat(nextBeatInBar))
            val playAtBeat = nextSub == 0

            // Dormir jusqu'à ~2 ms avant la cible
            val sleepMs = ((nextSubInNs - 2_000_000L) / 1_000_000L).coerceAtLeast(1L)
            delay(sleepMs)
            while (System.nanoTime() < targetNs) { /* spin court */ }

            if (playAtBeat) {
                val pcm = when {
                    strong && accent1 -> bufferStrong
                    medium && (accent13 || clock.signature.beatsPerBar >= 4) -> bufferMedium
                    else -> bufferNormal
                }
                writeWithGain(pcm, volume)
            } else if (clock.subdivision != TransportClock.Subdivision.Quarter) {
                writeWithGain(bufferNormal, volume * 0.4f)
            }

            clock.emitTickFromScheduler(nextBar, nextBeatInBar, nextSub, strong, medium, targetNs)
            // Petite pause pour ne pas saturer le scheduler si bpm élevé
            delay(1L)
        }
    }

    private fun writeWithGain(pcm: ShortArray, gain: Float) {
        val track = audioTrack ?: return
        val scaled = ShortArray(pcm.size)
        val g = gain.coerceIn(0f, 1f)
        for (i in pcm.indices) scaled[i] = (pcm[i] * g).toInt().coerceIn(-32768, 32767).toShort()
        try { track.write(scaled, 0, scaled.size) } catch (_: Exception) {}
    }

    // ─── Synthèse sons (44.1 kHz mono) ───

    private fun synthStrong(s: Sound): ShortArray = synth(s, 1.0f, strong = true)
    private fun synthMedium(s: Sound): ShortArray = synth(s, 0.8f, strong = false)
    private fun synthNormal(s: Sound): ShortArray = synth(s, 0.55f, strong = false)

    private fun synth(sound: Sound, gain: Float, strong: Boolean): ShortArray = when (sound) {
        Sound.Woodblock -> synthWoodblock(gain, strong)
        Sound.Beep -> synthBeep(gain, strong)
        Sound.Digital -> synthDigital(gain, strong)
        Sound.Click -> synthClick(gain)
        Sound.Kick -> synthKick(gain)
        Sound.Conga -> synthConga(gain, strong)
    }

    private fun synthWoodblock(gain: Float, strong: Boolean): ShortArray {
        val dur = 0.035f
        val n = (sampleRate * dur).toInt()
        val out = ShortArray(n)
        val f0 = if (strong) 1050f else 900f
        val decay = 55f
        for (i in 0 until n) {
            val t = i.toFloat() / sampleRate
            val env = exp(-decay * t)
            val v = sin(2.0 * PI * f0 * t) * env + 0.3 * sin(2.0 * PI * f0 * 2.1 * t) * env
            out[i] = (v * gain * 0.85f * 32767f).toInt().toShort()
        }
        return out
    }

    private fun synthBeep(gain: Float, strong: Boolean): ShortArray {
        val dur = 0.028f
        val n = (sampleRate * dur).toInt()
        val out = ShortArray(n)
        val f0 = if (strong) 1200f else 1000f
        for (i in 0 until n) {
            val t = i.toFloat() / sampleRate
            val env = if (t < 0.002f) t / 0.002f else exp(-70f * (t - 0.002f))
            val v = sin(2.0 * PI * f0 * t) * env
            out[i] = (v * gain * 0.7f * 32767f).toInt().toShort()
        }
        return out
    }

    private fun synthDigital(gain: Float, strong: Boolean): ShortArray {
        val dur = 0.018f
        val n = (sampleRate * dur).toInt()
        val out = ShortArray(n)
        val f0 = if (strong) 1800f else 1400f
        for (i in 0 until n) {
            val t = i.toFloat() / sampleRate
            val env = exp(-110f * t)
            val v = if (sin(2.0 * PI * f0 * t) > 0) 1.0 else -1.0
            out[i] = (v * env * gain * 0.5f * 32767f).toInt().toShort()
        }
        return out
    }

    private fun synthClick(gain: Float): ShortArray {
        val dur = 0.008f
        val n = (sampleRate * dur).toInt()
        val out = ShortArray(n)
        for (i in 0 until n) {
            val t = i.toFloat() / sampleRate
            val env = exp(-240f * t)
            val noise = (Math.random() * 2 - 1).toFloat()
            out[i] = (noise * env * gain * 0.6f * 32767f).toInt().toShort()
        }
        return out
    }

    private fun synthKick(gain: Float): ShortArray {
        val dur = 0.18f
        val n = (sampleRate * dur).toInt()
        val out = ShortArray(n)
        var phase = 0.0
        for (i in 0 until n) {
            val t = i.toFloat() / sampleRate
            val env = exp(-25f * t)
            val f = 60f * (1f + exp(-30f * t) * 3f)
            phase += 2.0 * PI * f / sampleRate
            val v = sin(phase) * env
            out[i] = (v * gain * 0.9f * 32767f).toInt().coerceIn(-32768, 32767).toShort()
        }
        return out
    }

    private fun synthConga(gain: Float, strong: Boolean): ShortArray {
        val dur = 0.11f
        val n = (sampleRate * dur).toInt()
        val out = ShortArray(n)
        val f0 = if (strong) 420f else 380f
        for (i in 0 until n) {
            val t = i.toFloat() / sampleRate
            val env = exp(-25f * t)
            val v = sin(2.0 * PI * f0 * t) * env + 0.35 * sin(2.0 * PI * f0 * 2.0 * t) * env
            out[i] = (v * gain * 0.85f * 32767f).toInt().toShort()
        }
        return out
    }
}

package com.soundstage.mixer.audio

import kotlinx.coroutines.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

/**
 * Looper event-only, 8 cases, 4 couches par case. Synchronisé sur TransportClock.
 * Aucun son joué ici — les events sont exposés via StateFlow ; c'est le consumer
 * (DrumPadScreen via ViewModel) qui les rejoue en audio.
 *
 * Supporte : length Auto / 1 / 2 / 4 / 8 bars, quantize 1/16 ou Free,
 * swing 0-100 % appliqué aux pas impairs du cycle, count-in au prochain début
 * de mesure.
 */
class LooperEngine(private val clock: TransportClock) {

    enum class CaseState { Empty, Queued, Recording, Playing, Overdubbing, Stopped }

    data class RecordedEvent(
        val kind: String,       // "pad" ou "note"
        val idx: Int,           // padId ou index de note 0-11
        val color: String,      // couleur UI au moment de l'enregistrement
        val beatOffset: Double, // position dans le cycle en beats (0..bars*beatsPerBar)
    )

    data class Case(
        val id: Int,
        val state: CaseState = CaseState.Empty,
        val bars: Int = 1,
        val lengthMode: String = "auto",  // "auto" | "1" | "2" | "4" | "8"
        val layers: List<List<RecordedEvent>> = emptyList(),
        val pendingRecEvents: List<RecordedEvent> = emptyList(),
        val pendingOverdubEvents: List<RecordedEvent> = emptyList(),
        val queuedAction: String? = null, // "rec" | "overdub" | "play"
        val queuedAtNs: Long = 0L,
        val recStartNs: Long = 0L,
        val startAtNs: Long = 0L,
        val lastBeatIndex: Double = -1.0,
        val offset: Int = 0,
        val chokeGroup: Int = 0,
    )

    data class UiState(
        val cases: List<Case> = List(8) { Case(id = it) },
        val quantize: Boolean = true,
        val swing: Int = 54,
        val lengthMode: String = "auto",
        val isPlayingAny: Boolean = false,
    )

    private val _state = MutableStateFlow(UiState())
    val state: StateFlow<UiState> = _state.asStateFlow()

    private var schedulerJob: Job? = null
    private val scope = CoroutineScope(Dispatchers.Default + SupervisorJob())

    fun start() {
        if (schedulerJob?.isActive == true) return
        schedulerJob = scope.launch { runScheduler() }
    }

    fun stop() {
        schedulerJob?.cancel()
        schedulerJob = null
    }

    fun release() {
        stop()
        scope.cancel()
    }

    fun setQuantize(q: Boolean) = _state.update { it.copy(quantize = q) }
    fun setSwing(s: Int) = _state.update { it.copy(swing = s.coerceIn(0, 100)) }
    fun setLengthMode(m: String) = _state.update { it.copy(lengthMode = m) }

    /** Tap sur une case (empty → queued rec ; rec → finish ; play → queued overdub ; etc.). */
    fun tapCase(caseIndex: Int) {
        val nowNs = System.nanoTime()
        val s = _state.value
        val busy = s.cases.any { it.id != caseIndex && (it.state == CaseState.Recording || (it.state == CaseState.Queued && it.queuedAction == "rec")) }
        val c = s.cases[caseIndex]
        val next = when (c.state) {
            CaseState.Empty -> {
                if (busy) return
                val nextBarNs = nowNs + clock.timeToNextBar(nowNs)
                c.copy(state = CaseState.Queued, queuedAction = "rec", queuedAtNs = nextBarNs, lengthMode = s.lengthMode)
            }
            CaseState.Queued -> c.copy(state = CaseState.Empty, queuedAction = null)
            CaseState.Recording -> finishRecording(c, nowNs)
            CaseState.Playing -> {
                val loopDur = clock.barDurationNs() * c.bars
                val sinceStart = (nowNs - c.startAtNs).coerceAtLeast(0L)
                val nextBoundary = c.startAtNs + ((sinceStart / loopDur) + 1) * loopDur
                c.copy(state = CaseState.Queued, queuedAction = "overdub", queuedAtNs = nextBoundary)
            }
            CaseState.Overdubbing -> commitOverdub(c)
            CaseState.Stopped -> c.copy(state = CaseState.Queued, queuedAction = "play", queuedAtNs = nowNs + clock.timeToNextBar(nowNs))
        }
        _state.update { st -> st.copy(cases = st.cases.toMutableList().also { it[caseIndex] = next }) }
    }

    fun longPressCaseAction(caseIndex: Int, action: String) {
        val c = _state.value.cases[caseIndex]
        val next = when (action) {
            "stop" -> c.copy(state = CaseState.Stopped)
            "play" -> c.copy(state = CaseState.Queued, queuedAction = "play", queuedAtNs = System.nanoTime() + clock.timeToNextBar())
            "clear" -> Case(id = caseIndex)
            "undo" -> if (c.layers.isNotEmpty()) c.copy(layers = c.layers.dropLast(1).let { if (it.isEmpty()) c.copy(state = CaseState.Empty, layers = emptyList()) else c.copy(layers = it) }) else c
            "double" -> if (c.bars < 8 && c.layers.isNotEmpty()) c.copy(bars = c.bars * 2, layers = c.layers.map { l -> l + l.map { e -> e.copy(beatOffset = e.beatOffset + c.bars * clock.signature.beatsPerBar) } }) else c
            else -> c
        }
        _state.update { st -> st.copy(cases = st.cases.toMutableList().also { it[caseIndex] = next }) }
    }

    fun stopAll() {
        _state.update { st -> st.copy(cases = st.cases.map { if (it.state == CaseState.Playing || it.state == CaseState.Overdubbing) it.copy(state = CaseState.Stopped) else it }) }
    }

    fun playAll() {
        val now = System.nanoTime()
        val nextBar = now + clock.timeToNextBar(now)
        _state.update { st -> st.copy(cases = st.cases.map { if (it.state == CaseState.Stopped) it.copy(state = CaseState.Queued, queuedAction = "play", queuedAtNs = nextBar) else it }) }
    }

    /** Appelé par le consumer quand un pad ou une note est joué pendant un REC ou OVERDUB. */
    fun record(kind: String, idx: Int, color: String) {
        val nowNs = System.nanoTime()
        _state.update { s ->
            val updated = s.cases.map { c ->
                when (c.state) {
                    CaseState.Recording -> {
                        val beatsPerBar = clock.signature.beatsPerBar
                        val elapsedNs = (nowNs - c.recStartNs).coerceAtLeast(0L)
                        val beatOffset = elapsedNs.toDouble() / clock.beatDurationNs().toDouble()
                        val quantized = if (s.quantize) {
                            val subdiv = clock.subdivision.subdivPerBeat.toDouble()
                            (Math.round(beatOffset * subdiv) / subdiv)
                        } else beatOffset
                        c.copy(pendingRecEvents = c.pendingRecEvents + RecordedEvent(kind, idx, color, quantized))
                    }
                    CaseState.Overdubbing -> {
                        val beatsPerBar = clock.signature.beatsPerBar
                        val loopBeats = c.bars * beatsPerBar
                        val elapsedNs = (nowNs - c.startAtNs).coerceAtLeast(0L)
                        val cycleNs = clock.barDurationNs() * c.bars
                        val offsetInCycleNs = elapsedNs % cycleNs
                        var beatOffset = offsetInCycleNs.toDouble() / clock.beatDurationNs().toDouble()
                        beatOffset %= loopBeats
                        val quantized = if (s.quantize) {
                            val subdiv = clock.subdivision.subdivPerBeat.toDouble()
                            (Math.round(beatOffset * subdiv) / subdiv)
                        } else beatOffset
                        c.copy(pendingOverdubEvents = c.pendingOverdubEvents + RecordedEvent(kind, idx, color, quantized))
                    }
                    else -> c
                }
            }
            s.copy(cases = updated)
        }
    }

    private fun finishRecording(c: Case, nowNs: Long): Case {
        val elapsedNs = nowNs - c.recStartNs
        val bars = if (c.lengthMode == "auto") {
            (Math.round(elapsedNs.toDouble() / clock.barDurationNs().toDouble())).toInt().coerceAtLeast(1)
        } else c.lengthMode.toIntOrNull() ?: 1
        val layer = c.pendingRecEvents.map { e ->
            val loopBeats = bars * clock.signature.beatsPerBar
            e.copy(beatOffset = e.beatOffset % loopBeats)
        }
        val nextStartNs = c.recStartNs + bars * clock.barDurationNs()
        return c.copy(
            state = CaseState.Playing,
            bars = bars,
            layers = listOf(layer),
            pendingRecEvents = emptyList(),
            startAtNs = nextStartNs,
            queuedAction = null,
        )
    }

    private fun commitOverdub(c: Case): Case {
        if (c.layers.size >= 4) return c.copy(state = CaseState.Playing, pendingOverdubEvents = emptyList(), queuedAction = null)
        val loopBeats = c.bars * clock.signature.beatsPerBar
        val layer = c.pendingOverdubEvents.map { e -> e.copy(beatOffset = e.beatOffset % loopBeats) }
        return c.copy(
            state = CaseState.Playing,
            layers = c.layers + layer,
            pendingOverdubEvents = emptyList(),
            queuedAction = null,
        )
    }

    private suspend fun runScheduler() {
        while (schedulerJob?.isActive == true) {
            val nowNs = System.nanoTime()
            var changed = false
            val updated = _state.value.cases.map { c ->
                when (c.state) {
                    CaseState.Queued -> if (nowNs >= c.queuedAtNs) {
                        changed = true
                        when (c.queuedAction) {
                            "rec" -> c.copy(state = CaseState.Recording, recStartNs = c.queuedAtNs, pendingRecEvents = emptyList(), queuedAction = null)
                            "overdub" -> c.copy(state = CaseState.Overdubbing, startAtNs = c.queuedAtNs, pendingOverdubEvents = emptyList(), queuedAction = null)
                            "play" -> c.copy(state = CaseState.Playing, startAtNs = c.queuedAtNs, queuedAction = null, lastBeatIndex = -1.0)
                            else -> c
                        }
                    } else c
                    CaseState.Recording -> {
                        if (c.lengthMode != "auto") {
                            val targetBars = c.lengthMode.toIntOrNull() ?: 1
                            val endNs = c.recStartNs + targetBars * clock.barDurationNs()
                            if (nowNs >= endNs) { changed = true; finishRecording(c, endNs) } else c
                        } else c
                    }
                    CaseState.Overdubbing -> {
                        val endNs = c.startAtNs + c.bars * clock.barDurationNs()
                        if (nowNs >= endNs) { changed = true; commitOverdub(c) } else c
                    }
                    else -> c
                }
            }
            if (changed) _state.update { it.copy(cases = updated) }
            delay(15L)
        }
    }
}

package com.soundstage.mixer.model

/**
 * Modèle des 2 chaînes FX (Tonic et Pads), indépendantes.
 * Chaque paramètre est un Float 0..1 (UI 0..100 %).
 * L'ordre des effets est fixe et appliqué côté natif :
 *   Filter -> Brightness -> Delay -> Reverb -> Shimmer (Shimmer : Tonic only)
 */
data class SurfaceFxChain(
    val filterOn: Boolean = false,
    val filterAmount: Float = 0.5f,
    val brightnessOn: Boolean = false,
    val brightnessAmount: Float = 0.5f,
    val delayOn: Boolean = false,
    val delayAmount: Float = 0.3f,
    val reverbOn: Boolean = true,
    val reverbAmount: Float = 0.4f,
    val shimmerOn: Boolean = false,
    val shimmerAmount: Float = 0.3f,
) {
    fun enabledCount(includeShimmer: Boolean): Int {
        var n = 0
        if (filterOn) n++
        if (brightnessOn) n++
        if (delayOn) n++
        if (reverbOn) n++
        if (includeShimmer && shimmerOn) n++
        return n
    }
    fun withShimmerDisabled(): SurfaceFxChain = copy(shimmerOn = false, shimmerAmount = 0f)
}

data class FxChainsState(
    val tonic: SurfaceFxChain = SurfaceFxChain(reverbOn = true, reverbAmount = 0.4f, shimmerOn = true, shimmerAmount = 0.4f, brightnessOn = true, brightnessAmount = 0.6f),
    val pads: SurfaceFxChain = SurfaceFxChain(reverbOn = true, reverbAmount = 0.3f, brightnessOn = true, brightnessAmount = 0.5f),
) {
    companion object {
        val DEFAULT = FxChainsState()
    }
}

        val kitId = "kit_${System.currentTimeMillis()}"
        val samples = _uiState.value.drumPads
            .sortedBy { it.id }
            .map { it.sampleFileName.ifEmpty { null } }
        val newKit = com.soundstage.mixer.model.DrumKitData(id = kitId, name = cleanName, padSamples = samples)
        _uiState.update { it.copy(kits = it.kits + newKit, activeDrumKitName = cleanName) }
        persistCurrentStateDebounced()
    }

    fun loadKit(kitId: String) {
        val kit = _uiState.value.kits.find { it.id == kitId || it.name == kitId } ?: return
        _uiState.update { state ->
            val updatedPads = state.drumPads.mapIndexed { idx, pad ->
                if (idx < kit.padSamples.size) {
                    val sampleName = kit.padSamples[idx]
                    pad.copy(
                        label = sampleName?.take(12) ?: "Empty",
                        sampleFileName = sampleName ?: "",
                        sampleFilePath = "",
                    )
                } else pad
            }
            state.copy(drumPads = updatedPads, activeDrumKitName = kit.name)
        }
        persistCurrentStateDebounced()
    }

    fun deleteKit(kitId: String) {
        _uiState.update { state ->
            val remaining = state.kits.filterNot { it.id == kitId }
            val nextName = if (state.kits.find { it.id == kitId }?.name == state.activeDrumKitName) {
                remaining.firstOrNull()?.name ?: ""
            } else state.activeDrumKitName
            state.copy(kits = remaining, activeDrumKitName = nextName)
        }
        persistCurrentStateDebounced()
    }

    private fun buildKitsJson(kits: List<com.soundstage.mixer.model.DrumKitData>): String {
        return try {
            val arr = org.json.JSONArray()
            kits.forEach { kit ->
                val o = org.json.JSONObject().apply {
                    put("id", kit.id)
                    put("name", kit.name)
                    val s = org.json.JSONArray()
                    kit.padSamples.forEach { if (it == null) s.put(org.json.JSONObject.NULL) else s.put(it) }
                    put("padSamples", s)
                }
                arr.put(o)
            }
            arr.toString()
        } catch (_: Exception) { "" }
    }

    private fun parseKitsJson(json: String?): List<com.soundstage.mixer.model.DrumKitData> {
        if (json.isNullOrEmpty()) return emptyList()
        return try {
            val arr = org.json.JSONArray(json)
            val list = mutableListOf<com.soundstage.mixer.model.DrumKitData>()
            for (i in 0 until arr.length()) {
                val o = arr.getJSONObject(i)
                val sArr = o.optJSONArray("padSamples")
                val samples = mutableListOf<String?>()
                if (sArr != null) {
                    for (j in 0 until sArr.length()) {
                        if (sArr.isNull(j)) samples.add(null)
                        else samples.add(sArr.getString(j))
                    }
                }
                list.add(com.soundstage.mixer.model.DrumKitData(
                    id = o.optString("id", "kit_$i"),
                    name = o.optString("name", "Kit ${i + 1}"),
                    padSamples = samples,
                ))
            }
            list
        } catch (_: Exception) { emptyList() }
    }

// ─── Roll — animation de chiffre qui change ───
@Composable
fun DpRoll(value: Any, style: TextStyle) {
    val prev = remember { mutableStateOf(value) }
    var trigger by remember { mutableIntStateOf(0) }
    LaunchedEffect(value) {
        if (prev.value != value) {
            prev.value = value
            trigger++
        }
    }
    Text(
        text = value.toString(),
        style = style,
        modifier = Modifier.graphicsLayerAlphaOnChange(trigger)
    )
}

private fun Modifier.graphicsLayerAlphaOnChange(trigger: Int): Modifier {
    val alpha by animateFloatAsState(
        targetValue = 1f,
        animationSpec = tween(220),
        label = "roll$trigger"
    )
    return this.then(Modifier.graphicsLayerCompat(alpha))
}

private fun Modifier.graphicsLayerCompat(alpha: Float): Modifier =
    androidx.compose.ui.draw.alpha(alpha)

// ─── Seg — segmenté générique ───
@Composable
fun <T> DpSeg(
    options: List<Pair<T, String>>,
    value: T,
    onChange: (T) -> Unit,
    modifier: Modifier = Modifier,
    height: Dp = 22.dp,
    warn: Boolean = false,
) {
    val idx = options.indexOfFirst { it.first == value }.coerceAtLeast(0)
    val fraction by animateFloatAsState(
        targetValue = if (options.isNotEmpty()) idx.toFloat() / options.size else 0f,
        animationSpec = spring(dampingRatio = 0.72f, stiffness = 500f),
        label = "segFrac"
    )
    Box(
        modifier
            .height(height + 6.dp)
            .clip(RoundedCornerShape(12.dp))
            .background(Color(0xCC0F141D))
            .border(1.dp, dpWhite(0.08f), RoundedCornerShape(12.dp))
            .padding(3.dp)
    ) {
        if (options.isNotEmpty()) {
            val segFraction = 1f / options.size
            Box(
                Modifier
                    .fillMaxHeight()
                    .fillMaxWidth(segFraction)
                    .offset(x = with(androidx.compose.ui.platform.LocalDensity.current) {
                        (fraction * options.size * (0)).dp // placeholder, offset calculé par Layout
                    })
                    .clip(RoundedCornerShape(9.dp))
                    .background(Brush.verticalGradient(listOf(Color(0xFF38445E), Color(0xFF2B3650))))
            )
        }
        Row(Modifier.fillMaxSize()) {
            options.forEach { (key, label) ->
                val on = key == value
                val c by animateColorAsState(
                    if (on) (if (warn && label == "Edit") DpColors.Warm else Color(0xFFF1F4FA)) else DpColors.Mute,
                    tween(200),
                    label = "segColor"
                )
                Box(
                    Modifier
                        .weight(1f)
                        .fillMaxHeight()
                        .pointerInput(Unit) { detectTapGestures { onChange(key) } },
                    contentAlignment = Alignment.Center
                ) {
                    Text(label, style = dpText(11.5f, FontWeight.Bold, c), maxLines = 1)
                }
            }
        }
    }
}

// ─── LedRing — anneau LED (statique ou rotatif) ───
@Composable
fun DpLedRing(
    color: Color,
    mode: String,           // "off" | "run" | "queued" | "solid"
    durSeconds: Float = 2f,
    cornerDp: Dp = 18.dp,
    strokeDp: Dp = 2.5.dp,
) {
    if (mode == "off") return
    val infiniteTransition = rememberInfiniteTransition(label = "led")
    val rotation by if (mode == "run") {
        infiniteTransition.animateFloat(
            initialValue = 0f,
            targetValue = 360f,
            animationSpec = infiniteRepeatable(tween((durSeconds * 1000).toInt(), easing = androidx.compose.animation.core.LinearEasing), androidx.compose.animation.core.RepeatMode.Restart),
            label = "ledRot"
        )
    } else remember { mutableFloatStateOf(0f) }
    val alpha by if (mode == "queued") {
        infiniteTransition.animateFloat(
            initialValue = 0.35f, targetValue = 1f,
            animationSpec = infiniteRepeatable(tween(520), androidx.compose.animation.core.RepeatMode.Reverse),
            label = "ledPulse"
        )
    } else remember { mutableFloatStateOf(1f) }
    androidx.compose.foundation.Canvas(
        Modifier
            .fillMaxSize()
            .clip(RoundedCornerShape(cornerDp))
    ) {
        val strokePx = strokeDp.toPx()
        val inset = strokePx / 2f
        val cornerPx = cornerDp.toPx()
        if (mode == "run") {
            androidx.compose.ui.graphics.drawscope.rotate(rotation) {
                drawArc(
                    brush = Brush.sweepGradient(listOf(Color.Transparent, color, Color.Transparent)),
                    startAngle = 0f, sweepAngle = 360f, useCenter = false,
                    topLeft = Offset(inset, inset),
                    size = androidx.compose.ui.geometry.Size(size.width - strokePx, size.height - strokePx),
                    style = Stroke(width = strokePx, cap = StrokeCap.Round),
                    alpha = alpha
                )
            }
        } else {
            drawRoundRect(
                color = color,
                topLeft = Offset(inset, inset),
                size = androidx.compose.ui.geometry.Size(size.width - strokePx, size.height - strokePx),
                cornerRadius = androidx.compose.ui.geometry.CornerRadius(cornerPx),
                style = Stroke(width = strokePx),
                alpha = alpha
            )
        }
    }
}

// ─── VolumeOverlay — 14 barres, pad → amber → rose ───
@Composable
fun DpVolumeOverlay(value: Float, padColor: Color, modifier: Modifier = Modifier) {
    val N = 14
    val active = Math.round((value / 100f) * N)
    Box(
        modifier
            .clip(RoundedCornerShape(18.dp))
            .background(Color(0xC7060914)),
        contentAlignment = Alignment.Center
    ) {
        Column(horizontalAlignment = Alignment.CenterHorizontally) {
            Text(
                "${value.toInt()}%",
                style = dpText(22f, FontWeight.ExtraBold, Color(0xFFF1F4FA))
            )
            Spacer(Modifier.height(4.dp))
            Row(verticalAlignment = Alignment.Bottom) {
                for (i in 0 until N) {
                    val on = i < active
                    val h = (6 + (i.toFloat() / N) * 22).dp
                    val segCol = when {
                        (i.toFloat() / N) > 0.82f -> DpColors.Rose
                        (i.toFloat() / N) > 0.6f -> DpColors.Warm
                        else -> padColor
                    }
                    Box(
                        Modifier
                            .padding(horizontal = 1.dp)
                            .width(4.dp)
                            .height(h)
                            .clip(RoundedCornerShape(2.dp))
                            .background(if (on) segCol else dpWhite(0.12f))
                    )
                }
            }
        }
    }
}

// ─── VolumeKnob — 3D slider drag vertical ───
@Composable
fun DpVolumeKnob(
    value: Float,               // 0..100
    onChange: (Float) -> Unit,
    modifier: Modifier = Modifier,
    sizeDp: Dp = 52.dp,
    onDoubleTap: Float = 80f,
    label: Boolean = true,
) {
    var dragStart by remember { mutableStateOf<Pair<Float, Float>?>(null) }
    val angle = -135f + (value / 100f) * 270f
    Row(
        modifier = modifier,
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.spacedBy(10.dp)
    ) {
        Box(
            Modifier
                .size(sizeDp)
                .pointerInput(value) {
                    detectDragGestures(
                        onDragStart = { dragStart = it.y to value },
                        onDragEnd = { dragStart = null },
                        onDragCancel = { dragStart = null },
                        onDrag = { change, dy ->
                            change.consume()
                            val start = dragStart ?: return@detectDragGestures
                            val newValue = (start.second + (start.first - change.position.y) * 0.6f).coerceIn(0f, 100f)
                            onChange(newValue)
                        },
                    )
                }
                .pointerInput(Unit) { detectTapGestures(onDoubleTap = { onChange(onDoubleTap) }) }
        ) {
            androidx.compose.foundation.Canvas(Modifier.fillMaxSize()) {
                val cx = this.size.width / 2
                val cy = this.size.height / 2
                val outerR = this.size.minDimension / 2 * 0.95f
                val innerR = outerR * 0.78f
                // Outer ring
                drawCircle(
                    color = dpWhite(0.08f),
                    radius = outerR,
                    center = Offset(cx, cy),
                    style = Stroke(width = 3f)
                )
                // Arc progress
                drawArc(
                    color = DpColors.Accent,
                    startAngle = 135f,
                    sweepAngle = (value / 100f) * 270f,
                    useCenter = false,
                    topLeft = Offset(cx - outerR, cy - outerR),
                    size = androidx.compose.ui.geometry.Size(outerR * 2, outerR * 2),
                    style = Stroke(width = 3f, cap = StrokeCap.Round),
                )
                // Body
                drawCircle(color = Color(0xFF2C3550), radius = innerR, center = Offset(cx, cy))
                drawCircle(color = dpWhite(0.14f), radius = innerR, center = Offset(cx, cy), style = Stroke(width = 1f))
                // Indicator
                val rad = Math.toRadians((angle + 90f).toDouble())
                val endX = cx + (innerR * 0.85f * cos(rad)).toFloat()
                val endY = cy + (innerR * 0.85f * sin(rad)).toFloat()
                drawLine(DpColors.Accent, Offset(cx, cy), Offset(endX, endY), strokeWidth = 2.5f, cap = StrokeCap.Round)
            }
        }
        if (label) {
            Column {
                Text("VOLUME", style = dpText(9.5f, FontWeight.Bold, DpColors.Dim).copy(letterSpacing = 1.2.sp))
                Text("${value.toInt()}%", style = dpText(17f, FontWeight.ExtraBold, Color(0xFFF1F4FA)))
            }
        }
    }
}

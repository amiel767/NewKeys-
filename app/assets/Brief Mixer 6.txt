@Composable
fun SustainPedalButton(
    isActive: Boolean,
    onToggle: () -> Unit,
    panelTop: Color = Color(0xFF2B3348),
    panelBottom: Color = Color(0xFF1D2433),
    modifier: Modifier = Modifier
) {
    val accent = Color(0xFF8FB2E8)
    val angle by animateFloatAsState(
        targetValue = if (isActive) 10f else -8f,
        animationSpec = spring(dampingRatio = 0.55f, stiffness = 600f),
        label = "pedal"
    )
    val glowAlpha by animateFloatAsState(
        targetValue = if (isActive) 0.55f else 0f,
        animationSpec = tween(280),
        label = "pedalGlow"
    )

    Box(
        modifier = modifier
            .width(56.dp)
            .height(40.dp)
            .clip(RoundedCornerShape(12.dp))
            .background(
                if (isActive)
                    Brush.verticalGradient(listOf(Color(0x478FB2E8), Color(0x248FB2E8)))
                else
                    Brush.verticalGradient(listOf(panelTop, panelBottom))
            )
            .border(1.dp, Color(0x0DFFFFFF), RoundedCornerShape(12.dp))
            .clickable { onToggle() },
        contentAlignment = Alignment.Center
    ) {
        Canvas(Modifier.fillMaxSize()) {
            val cx = size.width / 2f
            val baseY = size.height * 0.78f

            if (glowAlpha > 0.01f) {
                drawCircle(
                    color = accent.copy(alpha = glowAlpha * 0.45f),
                    radius = size.width * 0.55f,
                    center = Offset(cx, baseY + 6.dp.toPx())
                )
            }
            drawOval(
                color = Color.Black.copy(alpha = 0.55f),
                topLeft = Offset(cx - 16.dp.toPx(), baseY + 5.dp.toPx()),
                size = Size(32.dp.toPx() + if (isActive) 4.dp.toPx() else 0f, 4.dp.toPx())
            )
            val cable = Path().apply {
                moveTo(cx - 18.dp.toPx(), baseY - 4.dp.toPx())
                quadraticBezierTo(
                    cx - 26.dp.toPx(), baseY - 6.dp.toPx(),
                    cx - 30.dp.toPx(), baseY - 14.dp.toPx()
                )
            }
            drawPath(cable, Color(0xFF2E3549),
                style = Stroke(1.6.dp.toPx(), cap = StrokeCap.Round))

            val spring = Path().apply {
                moveTo(cx - 14.dp.toPx(), baseY - 2.dp.toPx())
                quadraticBezierTo(cx - 11.dp.toPx(), baseY, cx - 11.dp.toPx(), baseY + 2.dp.toPx())
                quadraticBezierTo(cx - 11.dp.toPx(), baseY + 4.dp.toPx(), cx - 14.dp.toPx(), baseY + 2.dp.toPx())
            }
            drawPath(spring, Color(0xFF5A6379),
                style = Stroke(1.1.dp.toPx(), cap = StrokeCap.Round))

            drawRoundRect(
                color = Color(0xFF2E3549),
                topLeft = Offset(cx - 19.dp.toPx(), baseY),
                size = Size(10.dp.toPx(), 5.dp.toPx()),
                cornerRadius = CornerRadius(1.4.dp.toPx())
            )
            drawCircle(
                color = Color(0xFF1A2030),
                radius = 1.dp.toPx(),
                center = Offset(cx - 14.dp.toPx(), baseY + 2.5.dp.toPx())
            )

            rotate(
                degrees = angle,
                pivot = Offset(cx - 14.dp.toPx(), baseY - 6.dp.toPx())
            ) {
                drawRoundRect(
                    brush = Brush.verticalGradient(listOf(
                        Color(0xFF6E7799), Color(0xFF4B5373), Color(0xFF3F4766))),
                    topLeft = Offset(cx - 14.dp.toPx(), baseY - 16.dp.toPx()),
                    size = Size(36.dp.toPx(), 10.dp.toPx()),
                    cornerRadius = CornerRadius(2.dp.toPx())
                )
                drawLine(
                    color = Color(0x0FFFFFFF),
                    start = Offset(cx - 12.dp.toPx(), baseY - 13.dp.toPx()),
                    end = Offset(cx + 20.dp.toPx(), baseY - 13.dp.toPx()),
                    strokeWidth = 0.6.dp.toPx()
                )
                drawRoundRect(
                    color = if (isActive) accent else Color(0xFF5A6379),
                    topLeft = Offset(cx + 20.dp.toPx(), baseY - 14.5.dp.toPx()),
                    size = Size(2.dp.toPx(), 6.dp.toPx()),
                    cornerRadius = CornerRadius(0.8.dp.toPx())
                )
            }
        }
    }
}

package com.soundstage.mixer.ui.components.v2

@Composable
fun ManageSnapshotDialogV2(
    slotKey: String,
    currentName: String,
    onDismiss: () -> Unit,
    onSave: (String, String) -> Unit
) {
    var name by remember { mutableStateOf(currentName) }
    var recallMode by remember { mutableStateOf("crossfade") }
    val tM = Color(0xFFF1F4FA)
    val tG = Color(0xFF7B859A)

    Dialog(
        onDismissRequest = onDismiss,
        properties = DialogProperties(usePlatformDefaultWidth = false)
    ) {
        Box(
            Modifier
                .width(460.dp)
                .clip(RoundedCornerShape(20.dp))
                .background(Brush.verticalGradient(listOf(
                    Color(0xFF232C3E), Color(0xFF1A2230))))
                .border(1.dp, Color(0x1AFFFFFF), RoundedCornerShape(20.dp))
                .padding(20.dp)
        ) {
            Column {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Text(
                        "Manage Snapshot",
                        color = tM,
                        fontSize = 18.sp,
                        fontWeight = FontWeight.ExtraBold,
                        modifier = Modifier.weight(1f)
                    )
                    Box(
                        Modifier
                            .size(30.dp)
                            .clip(RoundedCornerShape(15.dp))
                            .background(Color(0x0FFFFFFF))
                            .clickable { onDismiss() },
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            "×",
                            color = Color(0xFF9AA5BA),
                            fontSize = 18.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }
                }
                Spacer(Modifier.height(4.dp))
                Text(
                    if (slotKey == "slot_default") "Default snapshot"
                    else "Snapshot · used in this session",
                    color = tG,
                    fontSize = 11.5.sp
                )
                Spacer(Modifier.height(16.dp))
                Text(
                    "NAME",
                    color = tG,
                    fontSize = 10.5.sp,
                    fontWeight = FontWeight.ExtraBold,
                    letterSpacing = 1.2.sp
                )
                Spacer(Modifier.height(6.dp))
                BasicTextField(
                    value = name,
                    onValueChange = { name = it.take(24) },
                    singleLine = true,
                    textStyle = TextStyle(
                        color = tM,
                        fontSize = 16.sp,
                        fontWeight = FontWeight.SemiBold
                    ),
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(44.dp)
                        .background(Color(0xFF0F141D), RoundedCornerShape(12.dp))
                        .border(1.dp, Color(0x668FB2E8), RoundedCornerShape(12.dp))
                        .padding(horizontal = 14.dp)
                        .wrapContentHeight(Alignment.CenterVertically)
                )
                Spacer(Modifier.height(16.dp))
                Text(
                    "RECALL MODE",
                    color = tG,
                    fontSize = 10.5.sp,
                    fontWeight = FontWeight.ExtraBold,
                    letterSpacing = 1.2.sp
                )
                Spacer(Modifier.height(8.dp))
                Row(
                    Modifier
                        .fillMaxWidth()
                        .height(36.dp)
                        .clip(RoundedCornerShape(12.dp))
                        .background(Color(0xFF0F141D))
                        .padding(3.dp)
                ) {
                    listOf("immediate", "crossfade").forEach { mode ->
                        val sel = recallMode == mode
                        Box(
                            Modifier
                                .weight(1f)
                                .fillMaxHeight()
                                .clip(RoundedCornerShape(9.dp))
                                .background(if (sel) Color(0xFF38445E) else Color.Transparent)
                                .clickable { recallMode = mode },
                            contentAlignment = Alignment.Center
                        ) {
                            Text(
                                mode.replaceFirstChar { it.uppercase() },
                                color = if (sel) Color(0xFF8FB2E8) else tG,
                                fontSize = 13.sp,
                                fontWeight = FontWeight.Bold
                            )
                        }
                    }
                }
                Spacer(Modifier.height(18.dp))
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.End,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Box(
                        Modifier
                            .height(38.dp)
                            .clip(RoundedCornerShape(12.dp))
                            .background(Color(0x0AFFFFFF))
                            .border(1.dp, Color(0x1FFFFFFF), RoundedCornerShape(12.dp))
                            .clickable { onDismiss() }
                            .padding(horizontal = 20.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            "Cancel",
                            color = Color(0xFFC7CFDD),
                            fontSize = 14.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }
                    Spacer(Modifier.width(10.dp))
                    Box(
                        Modifier
                            .height(38.dp)
                            .clip(RoundedCornerShape(12.dp))
                            .background(Brush.verticalGradient(listOf(
                                Color(0xFF9DBDEB), Color(0xFF7FA3DD))))
                            .clickable {
                                if (name.isNotBlank()) onSave(name.trim(), recallMode)
                            }
                            .padding(horizontal = 24.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            "Save",
                            color = Color(0xFF0F1726),
                            fontSize = 14.sp,
                            fontWeight = FontWeight.ExtraBold
                        )
                    }
                }
            }
        }
    }
}

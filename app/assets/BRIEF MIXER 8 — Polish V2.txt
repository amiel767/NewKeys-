@Composable
fun IconGearV2(size: Dp = 20.dp, color: Color = Color(0xFFC9D1E0)) {
    Canvas(Modifier.size(size)) {
        val w = this.size.width
        val h = this.size.height
        val stroke = 2.dp.toPx()
        val center = Offset(w / 2f, h / 2f)
        val outer = w * 0.4f
        val inner = w * 0.25f
        
        drawCircle(color, inner, center, style = Stroke(stroke))
        
        val teeth = 8
        for (i in 0 until teeth) {
            val angle = (i * 360f / teeth) * (Math.PI / 180f).toFloat()
            val startX = center.x + (inner + stroke) * Math.cos(angle)
            val startY = center.y + (inner + stroke) * Math.sin(angle)
            val endX = center.x + outer * Math.cos(angle)
            val endY = center.y + outer * Math.sin(angle)
            drawLine(color, Offset(startX, startY), Offset(endX, endY), strokeWidth = stroke, cap = StrokeCap.Round)
        }
    }
}

        Box(Modifier.width(20.dp).fillMaxHeight().clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) { onMinus() },

            Text("−", color = Color(0xFFE2E8F0), fontSize = 12.sp, fontWeight = FontWeight.ExtraBold)
        }
        Column(Modifier.padding(horizontal = 2.dp).defaultMinSize(minWidth = 28.dp),
            horizontalAlignment = Alignment.CenterHorizontally, verticalArrangement = Arrangement.Center) {
            Text(label, color = tG, fontSize = 7.sp, fontWeight = FontWeight.ExtraBold, lineHeight = 8.sp)

        }
        Box(Modifier.width(20.dp).fillMaxHeight().clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) { onPlus() },
            contentAlignment = Alignment.Center) {
            Text("+", color = Color(0xFFE2E8F0), fontSize = 12.sp, fontWeight = FontWeight.ExtraBold)
        }
    }
}

@Composable
fun ChannelStripV2(
    track: TrackChannel, isDimmed: Boolean, accentColor: Color,
    onVolumeChange: (Float) -> Unit, onPanChange: (Float) -> Unit,
    onSoloToggle: () -> Unit, onMuteToggle: () -> Unit,
    onTrackNameClick: () -> Unit, onFxClick: () -> Unit,
    onTrackSettingsClick: () -> Unit, onPowerToggle: () -> Unit,
    onDeleteTrack: () -> Unit, modifier: Modifier = Modifier
) {
    val level = if (isDimmed) 0f else maxOf(track.peakMeterL, track.peakMeterR).coerceIn(0f, 1f)
    val auraAlpha = (0.17f + level * 0.42f) * (if (isDimmed) 0.3f else 1f)

    Box(
        modifier = modifier.clip(RoundedCornerShape(14.dp)) // Fix: 14dp au lieu de 18
            .background(Brush.verticalGradient(listOf(Color(0xFF232C3D), Color(0xFF191F2B))))
            .border(1.dp, Color(0x0DFFFFFF), RoundedCornerShape(14.dp))
    ) {
        Canvas(Modifier.fillMaxSize()) {
            drawRect(Brush.radialGradient(
                colors = listOf(accentColor.copy(alpha = auraAlpha), Color.Transparent),
                center = Offset(size.width / 2f, size.height * 1.15f),
                radius = size.height * 0.75f))
            drawRect(Brush.verticalGradient(colors = listOf(
                Color.Transparent, Color.Transparent, accentColor.copy(alpha = auraAlpha * 0.55f))))
        }

        Column(Modifier.fillMaxSize().padding(horizontal = 4.dp, vertical = 6.dp)) {
            // Name Plate : 22dp hauteur, font 9.5sp
            Box(
                modifier = Modifier.fillMaxWidth().height(22.dp)
                    .clip(RoundedCornerShape(8.dp))
                    .background(Brush.verticalGradient(listOf(Color(0xFF232C3D), Color(0xFF191F2B))))
                    .border(1.dp, Color(0x12FFFFFF), RoundedCornerShape(8.dp))
                    .clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) { onTrackNameClick() },
                contentAlignment = Alignment.Center
            ) {
                val name = track.patchName.ifBlank { track.soundfontName.ifBlank { "—" } }
                Text(name, color = if (isDimmed) Color(0xFF8F9AB0) else Color(0xFFE9EDF5),
                    fontSize = if (name.length > 15) 8.5.sp else 9.5.sp,
                    fontWeight = FontWeight.ExtraBold, maxLines = 1, overflow = TextOverflow.Ellipsis)
            }
            Spacer(Modifier.height(6.dp))
            
            // Pan + S/M : hauteur 24dp
            Row(Modifier.fillMaxWidth().height(24.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.spacedBy(4.dp)) {
                MicroPanKnob(track.pan, onPanChange, !isDimmed, accentColor, Modifier.size(24.dp))

                KeyBtnV2("M", track.isMuted, true, accentColor, onMuteToggle, Modifier.weight(1f).fillMaxHeight())
            }
            Spacer(Modifier.height(4.dp))

            FaderSliderWithVuMeter(
                value = track.volume, peakMeterL = track.peakMeterL, peakMeterR = track.peakMeterR,

@Composable
private fun KeyBtnV2(
    label: String, isActive: Boolean, isMute: Boolean,
    accentColor: Color, onClick: () -> Unit, modifier: Modifier = Modifier
) {
    val bg = when {
        !isActive -> Brush.verticalGradient(listOf(Color(0xFF3B4560), Color(0xFF2D364D)))

        else      -> Brush.verticalGradient(listOf(accentColor.copy(alpha = 0.85f), accentColor.copy(alpha = 0.60f)))
    }
    val fg = when {
        !isActive -> Color(0xFFB8C2D6)
        isMute    -> Color(0xFFF2B0C0)
        else      -> Color.White
    }
    val glowColor = when {
        !isActive -> Color.Transparent
        isMute    -> Color(0x70C2587A)
        else      -> accentColor.copy(alpha = 0.44f)
    }
    val glowBlur = if (isActive) 10.dp else 0.dp

    Box(
        modifier = modifier.clip(RoundedCornerShape(7.dp))
            .background(bg)
            .border(1.dp, if (isActive) Color.White.copy(alpha = 0.18f) else Color.Transparent, RoundedCornerShape(7.dp))
            .shadow(glowBlur, RoundedCornerShape(7.dp), spotColor = glowColor, ambientColor = glowColor)
            .clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) { onClick() },
        contentAlignment = Alignment.Center
    ) { Text(label, color = fg, fontSize = 11.sp, fontWeight = FontWeight.ExtraBold) } // Fix: 11sp
}

@Composable
private fun SnapshotCaseV2(
    key: String, displayName: String, isActive: Boolean,
    isDeletable: Boolean, isFlex: Boolean,
    onClick: () -> Unit, onLongPress: () -> Unit, onDelete: () -> Unit
) {
    var dragY by remember { mutableFloatStateOf(0f) }
    val animatedY by animateFloatAsState(dragY, animationSpec = if (dragY > 0f) tween(0) else spring(0.6f, 500f), label = "snapY")
    val isSwiping = dragY > 30f
    val splitParts = remember(displayName) {
        val m = Regex("^Snapshot\\s+(.+)$", RegexOption.IGNORE_CASE).find(displayName)
        if (m != null) Pair("SNAPSHOT", m.groupValues[1]) else null
    }
    Box(modifier = Modifier
        .then(if (isFlex) Modifier.weight(1f) else Modifier.width(90.dp)) // Fix: 90dp
        .fillMaxHeight()
        .offset { IntOffset(0, animatedY.roundToInt()) }
        .clip(RoundedCornerShape(10.dp)) // Fix: 10dp
        .background(if (isActive) Brush.verticalGradient(listOf(Color(0xFF8FB2E8), Color(0xFF7FA3DD))) else Brush.verticalGradient(listOf(Color(0xFF283144), Color(0xFF1D2433))))
        .border(1.dp, if (isActive) Color(0x998FB2E8) else Color(0x0DFFFFFF), RoundedCornerShape(10.dp))
        .pointerInput(key) {
            detectVerticalDragGestures(
                onVerticalDrag = { change, delta ->
                    change.consume()
                    if (delta > 0 && isDeletable) dragY = (dragY + delta).coerceAtMost(60f)
                },
                onDragEnd = { if (dragY > 30f && isDeletable) onDelete(); dragY = 0f },
                onDragCancel = { dragY = 0f })
        }
        .pointerInput(key) {
            detectTapGestures(onTap = { if (!isSwiping) onClick() }, onLongPress = { onLongPress() })
        },
        contentAlignment = Alignment.Center) {
        if (splitParts != null) {
            Column(horizontalAlignment = Alignment.CenterHorizontally) {
                Text("SNAPSHOT", color = Color(0x99FFFFFF), fontSize = 7.sp, fontWeight = FontWeight.ExtraBold, letterSpacing = 1.sp, lineHeight = 8.sp)
                Text(splitParts.second, color = if (isActive) Color(0xFF0F1726) else Color(0xFFDDE3EE), fontSize = 11.sp, fontWeight = FontWeight.ExtraBold, lineHeight = 12.sp, maxLines = 1, overflow = TextOverflow.Ellipsis)
            }
        } else {
            Text(displayName, color = if (isActive) Color(0xFF0F1726) else Color(0xFFDDE3EE), fontSize = 11.sp, fontWeight = FontWeight.ExtraBold, maxLines = 1, overflow = TextOverflow.Ellipsis)
        }
    }
}

@Composable
private fun MasterSliderCompact(
    value: Float, onValueChange: (Float) -> Unit, peakL: Float, peakR: Float, onFxClick: () -> Unit
) {
    var localPeak by remember { mutableFloatStateOf(0f) }
    // Fix: rememberUpdatedState pour que le drag lise la bonne valeur
    val currentValue by rememberUpdatedState(value)

    LaunchedEffect(peakL, peakR) {
        val target = value * maxOf(peakL, peakR)
        if (target > localPeak) localPeak = target
    }
    
    Row(Modifier.fillMaxSize().padding(horizontal = 6.dp), verticalAlignment = Alignment.CenterVertically) {
        Column(Modifier.weight(1f), verticalArrangement = Arrangement.Center) {
            Row(horizontalArrangement = Arrangement.SpaceBetween, modifier = Modifier.fillMaxWidth()) {
                Text("MASTER", color = Color(0xFF7B859A), fontSize = 7.sp, fontWeight = FontWeight.ExtraBold, letterSpacing = 1.sp)
                Text("${(value * 100).roundToInt()}%",
                    color = if (value > 0.75f) Color(0xFFD8A24A) else Color(0xFFF1F4FA),
                    fontSize = 9.sp, fontWeight = FontWeight.ExtraBold, fontFamily = FontFamily.Monospace)
            }
            Box(Modifier.fillMaxWidth().height(18.dp).pointerInput(Unit) {
                detectHorizontalDragGestures { change, delta ->
                    change.consume()
                    onValueChange((currentValue + delta / size.width.toFloat()).coerceIn(0f, 1f)) // Fix: currentValue
                }
            }) {
                Box(Modifier.fillMaxWidth().height(4.dp).align(Alignment.Center).clip(RoundedCornerShape(2.dp)).background(Color(0xFF0E131C)))
                Box(Modifier.fillMaxWidth(value).height(4.dp).align(Alignment.CenterStart).clip(RoundedCornerShape(2.dp)).background(Brush.horizontalGradient(listOf(Color(0x8C8FB2E8), Color(0xFF8FB2E8)))))
                if (localPeak > 0.01f) {
                    Box(Modifier.fillMaxHeight().width(2.dp).offset(x = (localPeak * 100).dp).background(Color(0xFFA9C6F0), RoundedCornerShape(1.dp)))
                }
                Box(Modifier.offset { IntOffset((value * 100).roundToInt(), 0) }
                    .size(14.dp, 16.dp).align(Alignment.CenterStart)
                    .clip(RoundedCornerShape(4.dp))
                    .background(Brush.verticalGradient(listOf(Color(0xFF6E7799), Color(0xFF4B5373), Color(0xFF3F4766))))
                    .border(1.dp, Color(0x29FFFFFF), RoundedCornerShape(4.dp)),
                    contentAlignment = Alignment.Center) {
                    Box(Modifier.fillMaxWidth(0.6f).height(1.5.dp).background(Color(0xFF8FB2E8), RoundedCornerShape(1.dp)))
                }
            }
        }
        Spacer(Modifier.width(6.dp))
        // Fix: Ajout du bouton FX
        Box(Modifier.size(24.dp).clip(RoundedCornerShape(6.dp)).background(Color(0xFF283144))
            .clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) { onFxClick() },
            contentAlignment = Alignment.Center) {
            Text("FX", color = Color(0xFF8FB2E8), fontSize = 8.sp, fontWeight = FontWeight.ExtraBold)
        }
    }
}

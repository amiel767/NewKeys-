@Composable
private fun SnapshotCaseV2(
    key: String, displayName: String, isActive: Boolean,
    isDeletable: Boolean, isFlex: Boolean,
    onClick: () -> Unit, onLongPress: () -> Unit, onDelete: () -> Unit
) {
    var dragY by remember { mutableFloatStateOf(0f) }
    val animatedY by animateFloatAsState(dragY,
        animationSpec = if (dragY > 0f) tween(0) else spring(0.6f, 500f),
        label = "snapY")
    val isSwiping = dragY > 30f
    val splitParts = remember(displayName) {
        val m = Regex("^Snapshot\\s+(.+)$", RegexOption.IGNORE_CASE).find(displayName)
        if (m != null) Pair("SNAPSHOT", m.groupValues[1]) else null
    }
    Box(modifier = Modifier
        .then(if (isFlex) Modifier.weight(1f) else Modifier.width(100.dp))
        .fillMaxHeight()
        .offset { IntOffset(0, animatedY.roundToInt()) }
        .clip(RoundedCornerShape(14.dp))
        .background(if (isActive)
            Brush.verticalGradient(listOf(Color(0xFF8FB2E8), Color(0xFF7FA3DD)))
            else Brush.verticalGradient(listOf(Color(0xFF283144), Color(0xFF1D2433))))
        .border(1.dp,
            if (isActive) Color(0x998FB2E8) else Color(0x0DFFFFFF),
            RoundedCornerShape(14.dp))
        .pointerInput(key) {
            detectVerticalDragGestures(
                onVerticalDrag = { change, delta ->
                    change.consume()
                    if (delta > 0 && isDeletable)
                        dragY = (dragY + delta).coerceAtMost(60f)
                },
                onDragEnd = {
                    if (dragY > 30f && isDeletable) onDelete()
                    dragY = 0f
                },
                onDragCancel = { dragY = 0f })
        }
        .pointerInput(key) {
            detectTapGestures(
                onTap = { if (!isSwiping) onClick() },
                onLongPress = { onLongPress() })
        },
        contentAlignment = Alignment.Center) {
        if (splitParts != null) {
            Column(horizontalAlignment = Alignment.CenterHorizontally) {
                Text("SNAPSHOT", color = Color(0x99FFFFFF), fontSize = 8.sp,
                    fontWeight = FontWeight.ExtraBold, letterSpacing = 1.2.sp,
                    lineHeight = 9.sp)
                Text(splitParts.second,
                    color = if (isActive) Color(0xFF0F1726) else Color(0xFFDDE3EE),
                    fontSize = 13.sp, fontWeight = FontWeight.ExtraBold,
                    lineHeight = 14.sp, maxLines = 1, overflow = TextOverflow.Ellipsis)
            }
        } else {
            Text(displayName,
                color = if (isActive) Color(0xFF0F1726) else Color(0xFFDDE3EE),
                fontSize = 13.sp, fontWeight = FontWeight.ExtraBold,
                maxLines = 1, overflow = TextOverflow.Ellipsis)
        }
    }
}

@Composable
private fun MasterSliderCompact(
    value: Float, onValueChange: (Float) -> Unit,
    peakL: Float, peakR: Float
) {
    var localPeak by remember { mutableFloatStateOf(0f) }
    LaunchedEffect(peakL, peakR) {
        val target = value * maxOf(peakL, peakR)
        if (target > localPeak) localPeak = target
    }
    Column(Modifier.fillMaxSize().padding(horizontal = 8.dp),
        verticalArrangement = Arrangement.Center) {
        Row(horizontalArrangement = Arrangement.SpaceBetween,
            modifier = Modifier.fillMaxWidth()) {
            Text("MASTER", color = Color(0xFF7B859A), fontSize = 8.sp,
                fontWeight = FontWeight.ExtraBold, letterSpacing = 1.sp)
            Text("${(value * 100).roundToInt()}%",
                color = if (value > 0.75f) Color(0xFFD8A24A) else Color(0xFFF1F4FA),
                fontSize = 10.sp, fontWeight = FontWeight.ExtraBold,
                fontFamily = FontFamily.Monospace)
        }
        Box(Modifier.fillMaxWidth().height(20.dp).pointerInput(Unit) {
            androidx.compose.foundation.gestures.detectHorizontalDragGestures { change, delta ->
                change.consume()
                onValueChange((value + delta / size.width.toFloat()).coerceIn(0f, 1f))
            }
        }) {
            Box(Modifier.fillMaxWidth().height(6.dp).align(Alignment.Center)
                .clip(RoundedCornerShape(3.dp)).background(Color(0xFF0E131C)))
            Box(Modifier.fillMaxWidth(value).height(6.dp).align(Alignment.CenterStart)
                .clip(RoundedCornerShape(3.dp))
                .background(Brush.horizontalGradient(listOf(
                    Color(0x8C8FB2E8), Color(0xFF8FB2E8)))))
            if (localPeak > 0.01f) {
                Box(Modifier.fillMaxHeight().width(2.dp)
                    .offset(x = (localPeak * 118).dp)
                    .background(Color(0xFFA9C6F0), RoundedCornerShape(1.dp)))
            }
            Box(Modifier.offset { IntOffset((value * 118).roundToInt(), 0) }
                .size(18.dp, 20.dp).align(Alignment.CenterStart)
                .clip(RoundedCornerShape(6.dp))
                .background(Brush.verticalGradient(listOf(
                    Color(0xFF6E7799), Color(0xFF4B5373), Color(0xFF3F4766))))
                .border(1.dp, Color(0x29FFFFFF), RoundedCornerShape(6.dp)),
                contentAlignment = Alignment.Center) {
                Box(Modifier.fillMaxWidth(0.6f).height(2.dp)
                    .background(Color(0xFF8FB2E8), RoundedCornerShape(1.dp)))
            }
        }
    }
}

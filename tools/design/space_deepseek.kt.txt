package com.soundstage.mixer.space 

 * ÆTHER SPACE — reverb panel, faithful to AetherSpacePreview.jsx. Design stage: 891 × 411 dp, scaled to any container.
 *
 * BEHAVIOUR (short brief)
 * - Knobs: the thin ring + line is ours. Drag up/down = change (full range ≈ 150 dp), double tap = back to the preset value.
 * Colour and glow follow the value: low = dim muted ring, high = bright saturated ring with a wide glow.
 * - Big DECAY knob: drag = decay, tap = fire the space (ripples + energy burst). It is always alive (slow breathing),
 * and its energy rises on fire and falls with the decay time. Each preset carries one of 8 CIRCULAR designs:
 * Hall (thick rings), Chamber (rotating dashed rings), Room (nested rounded squares), Shimmer (rings of orbiting stars),
 * Plate (polar dot grid with a radial wave), Cathedral (rose-window petals), Spring (circular coil), Cloud (rings of soft glows).
 * - Changing preset: values, the 5-colour palette and the design slide (values and colours ease, designs cross-fade).
 * - PRESETS: dropdown (FACTORY then USER), ◀ ▶ step through the list. A "•" after the name = unsaved changes.
 * - SAVE: opens a small name field; the preset joins the USER list (with a × to delete it) and becomes the current one.
 * - Engine hook: SpaceController(onParam = { key, value -> ... }) gets every change (also on preset load, key by key).
 * onFire is called on tap of the big knob (audition). onUserPresets lets you persist the USER list.
 * - Resources: 4 vector drawables (res/drawable): ic_space_prev, ic_space_next, ic_space_dropdown, ic_space_close.
 * Optional: put a bold rounded font (Poppins / Futura-like) in SpaceFont below.
 */

data class SpaceParam(val key: String, val label: String, val min: Float, val max: Float, val unit: String, val decimals: Int) {
 fun norm(v: Float) = ((v - min) / (max - min)).coerceIn(0f, 1f)
 fun fromNorm(n: Float) = min + n.coerceIn(0f, 1f) * (max - min)
 fun format(v: Float) = "%.${decimals}f %s".format(v, unit)
}

data class SpaceUiState(
 val user: List<SpacePreset> = emptyList(), val presetId: String = "f0", val values: Map<String, Float> = SpacePresets.factory[0].values,
 val dropdownOpen: Boolean = false, val saveOpen: Boolean = false
) {
 val all: List<SpacePreset> get() = SpacePresets.factory + user
 val preset: SpacePreset get() = all.firstOrNull { it.id == presetId } ?: SpacePresets.factory[0]
 val modified: Boolean get() = SpaceSpec.all.any { abs((values[it.key] ?: 0f) - (preset.values[it.key] ?: 0f)) > 1e-6f }
}

class SpaceController(
 initialUser: List<SpacePreset> = emptyList(),
 private val onParam: (key: String, value: Float) -> Unit = { _, _ -> },
 private val onUserPresets: (List<SpacePreset>) -> Unit = {},
 val onFire: () -> Unit = {}
) {
 private val _state = MutableStateFlow(SpaceUiState(user = initialUser))
 val state: StateFlow<SpaceUiState> = _state.asStateFlow()

 fun setValue(key: String, v: Float) { _state.update { it.copy(values = it.values + (key to v)) }; onParam(key, v) }
 fun reset(key: String) { val v = _state.value.preset.values[key] ?: return; setValue(key, v) }
 fun load(id: String) {
 val p = _state.value.all.firstOrNull { it.id == id } ?: return
 _state.update { it.copy(presetId = p.id, values = p.values, dropdownOpen = false) }
 p.values.forEach { (k, v) -> onParam(k, v) }
 }
 fun step(d: Int) { val l = _state.value.all; val i = l.indexOfFirst { it.id == _state.value.presetId }.coerceAtLeast(0); load(l[(i + d + l.size) % l.size].id) }
 fun dropdown(open: Boolean) = _state.update { it.copy(dropdownOpen = open, saveOpen = false) }
 fun saveDialog(open: Boolean) = _state.update { it.copy(saveOpen = open, dropdownOpen = false) }
 fun save(name: String) {
 val s = _state.value; val base = s.preset
 val p = SpacePreset("u" + System.currentTimeMillis(), name.trim().ifEmpty { "User ${s.user.size + 1}" }, base.style, base.palette, s.values, user = true)
 _state.update { it.copy(user = it.user + p, presetId = p.id, saveOpen = false) }
 onUserPresets(_state.value.user)
 }
 fun deleteUser(id: String) {
 val wasCurrent = _state.value.presetId == id
 _state.update { it.copy(user = it.user.filter { p -> p.id != id }) }
 onUserPresets(_state.value.user)
 if (wasCurrent) load("f0")
 }
}

private fun rememberSpaceTime(): Float {
 var t by remember { mutableFloatStateOf(0f) }
 LaunchedEffect(Unit) { while (true) withInfiniteAnimationFrameNanos { ns -> t = ns / 1e9f } }
 return t
}

@Composable
fun SpaceScreen(state: SpaceUiState, controller: SpaceController, modifier: Modifier = Modifier) {
 BoxWithConstraints(modifier) {
 val sc = min(maxWidth.value / 891f, maxHeight.value / 411f)
 Box(Modifier.size((891 * sc).dp, (411 * sc).dp).align(Alignment.Center).clip(RoundedCornerShape((20 * sc).dp))) {
 Box(Modifier.requiredSize(891.dp, 411.dp).layout { m, _ -> val p = m.measure(androidx.compose.ui.unit.Constraints.fixed(891.dp.roundToPx(), 411.dp.roundToPx())); layout((891 * sc * density).toInt(), (411 * sc * density).toInt()) { p.placeWithLayer(0, 0) { scaleX = sc; scaleY = sc; transformOrigin = TransformOrigin(0f, 0f) } } }) {
 SpaceStage(state, controller)
 }
 }
 }
}
